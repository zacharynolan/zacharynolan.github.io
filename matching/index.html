Gale-Shapley Matching Demo

Assigning PhD students to faculty for GA roles

Drag to reorder each ranking (top = most preferred). Everything updates live — change the preset, slider, or proposing side and the match recomputes immediately. Click Regenerate to reroll the random seed.

// Seed re-rolls only when Regenerate is clicked. Preset / correlation
// changes reuse the same seed so results are deterministic per seed.
seed = {
  regenClick;
  return Math.floor(Math.random() * 1e9);
}
// Generate initial preferences from preset. Subscribes to preset,
// correlation, and seed so any change triggers a fresh regeneration.
// Returns {faculty: [[...students]], students: [[...faculty]]}
initialPrefs = {
  const lcg = d3.randomLcg(seed / 1e9);
  const r = {
    uniform: lcg,
    normal: d3.randomNormal.source(lcg)(0, 1)
  };
  const mode = preset;
  const rho = correlation;

  // Permutation helper
  const shuffle = (arr) => {
    const a = [...arr];
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(r.uniform() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  };
  const range = Array.from({length: n}, (_, i) => i);

  // Qualities: faculty assess students on a shared latent score; students also assess faculty.
  const studentQuality = Array.from({length: n}, () => r.normal());
  const facultyQuality = Array.from({length: n}, () => r.normal());

  const rankBy = (scores) => {
    return range
      .slice()
      .sort((a, b) => scores[b] - scores[a]); // highest score first
  };

  let faculty, students;
  if (mode === "Random") {
    faculty = Array.from({length: n}, () => shuffle(range));
    students = Array.from({length: n}, () => shuffle(range));
  } else if (mode === "Identity (i ↔ i)") {
    faculty = Array.from({length: n}, (_, f) => {
      const rest = range.filter((x) => x !== f);
      return [f, ...rest];
    });
    students = Array.from({length: n}, (_, s) => {
      const rest = range.filter((x) => x !== s);
      return [s, ...rest];
    });
  } else {
    // Correlated / anti-correlated: each faculty ranks students by
    // studentQuality + noise; lower ρ → more noise → more idiosyncrasy.
    const sign = mode.startsWith("Anti-correlated") ? -1 : 1;
    const noiseSd = Math.max(0.01, Math.sqrt(1 - rho * rho) / Math.max(rho, 0.01));
    faculty = Array.from({length: n}, () => {
      const scores = studentQuality.map((q) => sign * q + noiseSd * r.normal());
      return rankBy(scores);
    });
    students = Array.from({length: n}, () => {
      const scores = facultyQuality.map((q) => sign * q + noiseSd * r.normal());
      return rankBy(scores);
    });
  }
  return {faculty, students};
}
// Build the draggable preference editor. Takes an initial ordering,
// returns an element whose `.value` is the current (reactive) ordering.
function prefEditor(initial, selfPrefix, otherPrefix) {
  const container = htl.html`<div class="pref-grid"></div>`;
  const state = initial.map((row) => [...row]);

  const rankLabels = (list) => {
    [...list.children].forEach((li, i) => {
      const rankEl = li.querySelector(".rank");
      if (rankEl) rankEl.textContent = (i + 1) + ".";
    });
  };

  state.forEach((row, idx) => {
    const panel = htl.html`<div class="pref-panel">
      <div class="pref-header">${selfPrefix}${idx + 1}</div>
      <ol class="pref-list"></ol>
    </div>`;
    const list = panel.querySelector(".pref-list");
    row.forEach((otherIdx) => {
      const li = htl.html`<li data-id="${otherIdx}"><span class="rank"></span>${otherPrefix}${otherIdx + 1}</li>`;
      list.appendChild(li);
    });
    rankLabels(list);
    // eslint-disable-next-line no-undef
    new Sortable(list, {
      animation: 120,
      ghostClass: "sortable-ghost",
      chosenClass: "sortable-chosen",
      onEnd: () => {
        state[idx] = [...list.children].map((li) => +li.dataset.id);
        rankLabels(list);
        container.value = state.map((r) => [...r]);
        container.dispatchEvent(new Event("input", {bubbles: true}));
      }
    });
    container.appendChild(panel);
  });

  container.value = state.map((r) => [...r]);
  return container;
}

function galeShapley(proposerPrefs, receiverPrefs) {
  const n = proposerPrefs.length;
  const receiverRank = receiverPrefs.map((row) => {
    const rk = new Array(n);
    row.forEach((p, i) => (rk[p] = i));
    return rk;
  });
  const nextProposal = new Array(n).fill(0);
  const engagedTo = new Array(n).fill(-1); // receiver idx -> proposer idx
  const free = Array.from({length: n}, (_, i) => i);
  while (free.length) {
    const p = free.shift();
    const r = proposerPrefs[p][nextProposal[p]++];
    const current = engagedTo[r];
    if (current === -1) {
      engagedTo[r] = p;
    } else if (receiverRank[r][p] < receiverRank[r][current]) {
      engagedTo[r] = p;
      free.push(current);
    } else {
      free.push(p);
    }
  }
  return engagedTo;
}
matching = {
  const students = studentPrefs;
  const faculty = facultyPrefs;
  const studentsPropose = proposerSide === "Students";
  const proposerPrefs = studentsPropose ? students : faculty;
  const receiverPrefs = studentsPropose ? faculty : students;
  const engagedTo = galeShapley(proposerPrefs, receiverPrefs);

  // Normalize into faculty -> student mapping regardless of proposer side.
  const facultyToStudent = new Array(n);
  if (studentsPropose) {
    // engagedTo is receiver(faculty) -> proposer(student)
    for (let f = 0; f < n; f++) facultyToStudent[f] = engagedTo[f];
  } else {
    // engagedTo is receiver(student) -> proposer(faculty)
    for (let s = 0; s < n; s++) facultyToStudent[engagedTo[s]] = s;
  }

  // Build pair records with each side's rank of its match.
  const facultyRankOf = faculty.map((row) => {
    const rk = new Array(n);
    row.forEach((s, i) => (rk[s] = i));
    return rk;
  });
  const studentRankOf = students.map((row) => {
    const rk = new Array(n);
    row.forEach((f, i) => (rk[f] = i));
    return rk;
  });
  const pairs = facultyToStudent.map((s, f) => ({
    faculty: f,
    student: s,
    facultyRankOfStudent: facultyRankOf[f][s] + 1,
    studentRankOfFaculty: studentRankOf[s][f] + 1
  }));
  const avg = (arr) => arr.reduce((a, b) => a + b, 0) / arr.length;
  return {
    pairs,
    avgFacultyRank: avg(pairs.map((p) => p.facultyRankOfStudent)),
    avgStudentRank: avg(pairs.map((p) => p.studentRankOfFaculty)),
    firstChoiceFaculty: pairs.filter((p) => p.facultyRankOfStudent === 1).length,
    firstChoiceStudents: pairs.filter((p) => p.studentRankOfFaculty === 1).length
  };
}
bipartiteDiagram = {
  const width = 760;
  const rowH = 34;
  const height = n * rowH + 40;
  const leftX = 170;
  const rightX = width - 170;
  const topPad = 20;
  const pairs = matching.pairs;

  const svg = d3.create("svg")
    .attr("width", width)
    .attr("height", height)
    .style("font", "13px system-ui, sans-serif");

  // Column headers
  svg.append("text")
    .attr("x", leftX).attr("y", 14)
    .attr("text-anchor", "end")
    .attr("font-weight", 600)
    .text("Faculty");
  svg.append("text")
    .attr("x", rightX).attr("y", 14)
    .attr("text-anchor", "start")
    .attr("font-weight", 600)
    .text("Student");

  // Nodes
  for (let i = 0; i < n; i++) {
    const y = topPad + i * rowH + rowH / 2;
    svg.append("circle").attr("cx", leftX).attr("cy", y).attr("r", 5).attr("fill", "#1e40af");
    svg.append("text").attr("x", leftX - 10).attr("y", y + 4)
      .attr("text-anchor", "end").text(`F${i + 1}`);
    svg.append("circle").attr("cx", rightX).attr("cy", y).attr("r", 5).attr("fill", "#166534");
    svg.append("text").attr("x", rightX + 10).attr("y", y + 4)
      .attr("text-anchor", "start").text(`S${i + 1}`);
  }

  // Color by average rank across both sides (1 = both got top choice → green; n = both worst → red).
  const edgeColor = (p) => {
    const avgRank = (p.facultyRankOfStudent + p.studentRankOfFaculty) / 2;
    const t = (avgRank - 1) / (n - 1);
    return d3.interpolateRdYlGn(1 - t);
  };

  pairs.forEach((p) => {
    const y1 = topPad + p.faculty * rowH + rowH / 2;
    const y2 = topPad + p.student * rowH + rowH / 2;
    svg.append("line")
      .attr("x1", leftX + 6).attr("y1", y1)
      .attr("x2", rightX - 6).attr("y2", y2)
      .attr("stroke", edgeColor(p))
      .attr("stroke-width", 2);
  });

  return svg.node();
}
