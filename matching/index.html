Matching Demo (Gale-Shapley Inspired)

Assigning PhD students to faculty for GA roles

Setup

  • 12 faculty (F1–F12), each requesting 20 GA hours/year — modelled as one 10-hour slot per semester.
  • 8 students (A–H), each providing 40 GA hours/year — modelled as two 10-hour slots per semester.
  • Some students are committed before matching for teaching or lab manager roles. Use the role selectors below to assign each role by semester, or set to “—” to leave the role uncovered. A student assigned to a role gives up both of that semester’s slots.
  • With all four roles assigned (defaults: A→Fall teaching, B→Spring teaching, C→Fall lab, D→Spring lab), 80 hours are removed and the remaining 240 hours (= 12 faculty × 20h) match the available student-slots exactly per semester.

Drag to reorder each ranking (top = most preferred). Everything updates live — change the preset, slider, or proposing side and the match recomputes immediately. Click Regenerate to reroll the random seed.

// Reactive: each role selector contributes a (student, semester) block.
// Same student picked twice in one semester is fine — both their slots
// are blocked once.
blocks = {
  const result = {};
  const add = (student, sem) => {
    if (student === "—") return;
    if (!result[student]) result[student] = [];
    if (!result[student].includes(sem)) result[student].push(sem);
  };
  add(teachingFall, "Fall");
  add(teachingSpring, "Spring");
  add(labFall, "Fall");
  add(labSpring, "Spring");
  return result;
}

// Generate initial preferences: 12 faculty × 8 students, 8 students × 12 faculty.
initialPrefs = {
  const lcg = d3.randomLcg(seed / 1e9);
  const r = {
    uniform: lcg,
    normal: d3.randomNormal.source(lcg)(0, 1)
  };
  const mode = preset;
  const rho = correlation;

  const shuffle = (arr) => {
    const a = [...arr];
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(r.uniform() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  };
  const studentRange = Array.from({length: nStudents}, (_, i) => i);
  const facultyRange = Array.from({length: nFaculty}, (_, i) => i);

  const studentQuality = Array.from({length: nStudents}, () => r.normal());
  const facultyQuality = Array.from({length: nFaculty}, () => r.normal());

  const rankBy = (range, scores) =>
    range.slice().sort((a, b) => scores[b] - scores[a]);

  let faculty, students;
  if (mode === "Random") {
    faculty = Array.from({length: nFaculty}, () => shuffle(studentRange));
    students = Array.from({length: nStudents}, () => shuffle(facultyRange));
  } else if (mode.startsWith("Mutual")) {
    // Each (faculty, student) pair shares a latent "fit" score. Both
    // sides rank by that same score plus per-evaluation noise — so a
    // faculty's ranking of S correlates with S's ranking of that
    // faculty. ρ → 1: rankings near-perfectly mutual; ρ → 0: random.
    const noiseSd = Math.max(0.01, Math.sqrt(1 - rho * rho) / Math.max(rho, 0.01));
    const fit = Array.from({length: nFaculty}, () =>
      Array.from({length: nStudents}, () => r.normal())
    );
    faculty = Array.from({length: nFaculty}, (_, f) => {
      const scores = studentRange.map((s) => fit[f][s] + noiseSd * r.normal());
      return rankBy(studentRange, scores);
    });
    students = Array.from({length: nStudents}, (_, s) => {
      const scores = facultyRange.map((f) => fit[f][s] + noiseSd * r.normal());
      return rankBy(facultyRange, scores);
    });
  } else {
    const sign = mode.startsWith("Anti-correlated") ? -1 : 1;
    const noiseSd = Math.max(0.01, Math.sqrt(1 - rho * rho) / Math.max(rho, 0.01));
    faculty = Array.from({length: nFaculty}, () => {
      const scores = studentQuality.map((q) => sign * q + noiseSd * r.normal());
      return rankBy(studentRange, scores);
    });
    students = Array.from({length: nStudents}, () => {
      const scores = facultyQuality.map((q) => sign * q + noiseSd * r.normal());
      return rankBy(facultyRange, scores);
    });
  }
  return {faculty, students};
}
// Build the draggable preference editor. Takes an initial ordering and
// label functions for panel headers and item labels; returns an element
// whose `.value` is the current (reactive) ordering.
function prefEditor(initial, panelLabel, itemLabel) {
  const container = htl.html`<div class="pref-grid"></div>`;
  container.style.setProperty("--cols", initial.length);
  const state = initial.map((row) => [...row]);

  const rankLabels = (list) => {
    [...list.children].forEach((li, i) => {
      const rankEl = li.querySelector(".rank");
      if (rankEl) rankEl.textContent = (i + 1) + ".";
    });
  };

  state.forEach((row, idx) => {
    const panel = htl.html`<div class="pref-panel">
      <div class="pref-header">${panelLabel(idx)}</div>
      <ol class="pref-list"></ol>
    </div>`;
    const list = panel.querySelector(".pref-list");
    row.forEach((otherIdx) => {
      const li = htl.html`<li data-id="${otherIdx}"><span class="rank"></span>${itemLabel(otherIdx)}</li>`;
      list.appendChild(li);
    });
    rankLabels(list);
    // eslint-disable-next-line no-undef
    new Sortable(list, {
      animation: 120,
      ghostClass: "sortable-ghost",
      chosenClass: "sortable-chosen",
      onEnd: () => {
        state[idx] = [...list.children].map((li) => +li.dataset.id);
        rankLabels(list);
        container.value = state.map((r) => [...r]);
        container.dispatchEvent(new Event("input", {bubbles: true}));
      }
    });
    container.appendChild(panel);
  });

  container.value = state.map((r) => [...r]);
  return container;
}

Resulting match

// Many-to-many deferred-acceptance Gale-Shapley supporting per-side
// capacities. Faculty have capacity 1 per semester; eligible students
// have capacity 2 per semester. Returns proposer- and receiver-side
// match arrays keyed by id.
function manyToManyGS(proposers, receivers, proposerPrefs, receiverPrefs, proposerCap, receiverCap) {
  const receiverRank = {};
  for (const r of receivers) {
    const rk = {};
    receiverPrefs[r].forEach((p, i) => { rk[p] = i; });
    receiverRank[r] = rk;
  }
  const nextProposalIdx = {};
  const proposerMatches = {};
  for (const p of proposers) {
    nextProposalIdx[p] = 0;
    proposerMatches[p] = [];
  }
  const receiverMatches = {};
  for (const r of receivers) receiverMatches[r] = [];

  const free = [...proposers];
  while (free.length) {
    const p = free.shift();
    if (proposerMatches[p].length >= proposerCap[p]) continue;
    if (nextProposalIdx[p] >= proposerPrefs[p].length) continue;
    const r = proposerPrefs[p][nextProposalIdx[p]++];

    if (receiverMatches[r].length < receiverCap[r]) {
      receiverMatches[r].push(p);
      proposerMatches[p].push(r);
    } else {
      const worst = receiverMatches[r].reduce((acc, q) =>
        receiverRank[r][q] > receiverRank[r][acc] ? q : acc
      );
      if (receiverRank[r][p] < receiverRank[r][worst]) {
        receiverMatches[r] = receiverMatches[r].filter((q) => q !== worst);
        proposerMatches[worst] = proposerMatches[worst].filter((x) => x !== r);
        receiverMatches[r].push(p);
        proposerMatches[p].push(r);
        if (proposerMatches[worst].length < proposerCap[worst] &&
            nextProposalIdx[worst] < proposerPrefs[worst].length) {
          free.push(worst);
        }
      }
    }

    if (proposerMatches[p].length < proposerCap[p] &&
        nextProposalIdx[p] < proposerPrefs[p].length) {
      free.push(p);
    }
  }
  return {proposerMatches, receiverMatches};
}
matching = {
  const studentsPropose = proposerSide === "Students";
  const facultyRange = Array.from({length: nFaculty}, (_, i) => i);
  const result = {};

  for (const sem of semesters) {
    const eligible = eligibleStudents(sem);
    const eligibleSet = new Set(eligible);
    // Faculty rankings are over all 8 students; per-semester we drop blocked ones.
    const facultyPrefsFiltered = facultyPrefs.map((row) =>
      row.filter((s) => eligibleSet.has(s))
    );

    let proposers, receivers, proposerPrefs, receiverPrefs, proposerCap, receiverCap;
    if (studentsPropose) {
      proposers = eligible;
      receivers = facultyRange;
      proposerPrefs = {};
      proposerCap = {};
      eligible.forEach((s) => {
        proposerPrefs[s] = studentPrefs[s];
        proposerCap[s] = 2;
      });
      receiverPrefs = {};
      receiverCap = {};
      facultyRange.forEach((f) => {
        receiverPrefs[f] = facultyPrefsFiltered[f];
        receiverCap[f] = 1;
      });
    } else {
      proposers = facultyRange;
      receivers = eligible;
      proposerPrefs = {};
      proposerCap = {};
      facultyRange.forEach((f) => {
        proposerPrefs[f] = facultyPrefsFiltered[f];
        proposerCap[f] = 1;
      });
      receiverPrefs = {};
      receiverCap = {};
      eligible.forEach((s) => {
        receiverPrefs[s] = studentPrefs[s];
        receiverCap[s] = 2;
      });
    }

    const m = manyToManyGS(
      proposers, receivers, proposerPrefs, receiverPrefs, proposerCap, receiverCap
    );

    const facultyToStudent = {};
    if (studentsPropose) {
      for (const f of facultyRange) facultyToStudent[f] = m.receiverMatches[f][0];
    } else {
      for (const f of facultyRange) facultyToStudent[f] = m.proposerMatches[f][0];
    }

    const pairs = [];
    for (const f of facultyRange) {
      const s = facultyToStudent[f];
      if (s === undefined) continue;
      pairs.push({
        faculty: f,
        student: s,
        facultyRankOfStudent: facultyPrefs[f].indexOf(s) + 1,
        studentRankOfFaculty: studentPrefs[s].indexOf(f) + 1
      });
    }
    const avg = (arr) => (arr.length ? arr.reduce((a, b) => a + b, 0) / arr.length : 0);

    result[sem] = {
      facultyToStudent,
      eligible,
      pairs,
      avgFacultyRank: avg(pairs.map((p) => p.facultyRankOfStudent)),
      avgStudentRank: avg(pairs.map((p) => p.studentRankOfFaculty)),
      firstChoiceFaculty: pairs.filter((p) => p.facultyRankOfStudent === 1).length,
      firstChoiceStudents: pairs.filter((p) => p.studentRankOfFaculty === 1).length
    };
  }
  return result;
}
function bipartiteForSemester(semData, semName) {
  const width = 480;
  const facRowH = 30;
  const facCount = nFaculty;
  const eligible = semData.eligible;
  const stuCount = Math.max(eligible.length, 1);
  const totalH = facCount * facRowH;
  const height = totalH + 40;
  const leftX = 130;
  const rightX = width - 130;
  const topPad = 20;

  const facY = (i) => topPad + i * facRowH + facRowH / 2;
  const stuY = (j) => topPad + (j + 0.5) * (totalH / stuCount);

  const stuPos = {};
  eligible.forEach((s, j) => { stuPos[s] = j; });

  const svg = d3.create("svg")
    .attr("viewBox", `0 0 ${width} ${height}`)
    .attr("preserveAspectRatio", "xMidYMid meet")
    .style("width", "100%")
    .style("height", "auto")
    .style("max-width", `${width}px`)
    .style("font", "13px system-ui, sans-serif");

  svg.append("text").attr("x", leftX).attr("y", 14)
    .attr("text-anchor", "end").attr("font-weight", 600)
    .text(`${semName} — Faculty`);
  svg.append("text").attr("x", rightX).attr("y", 14)
    .attr("text-anchor", "start").attr("font-weight", 600)
    .text(`${semName} — Student`);

  for (let i = 0; i < facCount; i++) {
    const y = facY(i);
    svg.append("circle").attr("cx", leftX).attr("cy", y).attr("r", 5).attr("fill", "#1e40af");
    svg.append("text").attr("x", leftX - 10).attr("y", y + 4)
      .attr("text-anchor", "end").text(`F${i + 1}`);
  }
  for (let j = 0; j < stuCount; j++) {
    const y = stuY(j);
    svg.append("circle").attr("cx", rightX).attr("cy", y).attr("r", 5).attr("fill", "#166534");
    svg.append("text").attr("x", rightX + 10).attr("y", y + 4)
      .attr("text-anchor", "start").text(studentNames[eligible[j]]);
  }

  // Color by mean of normalized ranks across both sides:
  // 0 = both got top → green; 1 = both got worst → red.
  const edgeColor = (p) => {
    const facNorm = (p.facultyRankOfStudent - 1) / (nStudents - 1);
    const stuNorm = (p.studentRankOfFaculty - 1) / (nFaculty - 1);
    const t = (facNorm + stuNorm) / 2;
    return d3.interpolateRdYlGn(1 - t);
  };

  semData.pairs.forEach((p) => {
    const y1 = facY(p.faculty);
    const y2 = stuY(stuPos[p.student]);
    svg.append("line")
      .attr("x1", leftX + 6).attr("y1", y1)
      .attr("x2", rightX - 6).attr("y2", y2)
      .attr("stroke", edgeColor(p))
      .attr("stroke-width", 2);
  });

  return svg.node();
}
function matchTableForSem(semData) {
  const rows = semData.pairs
    .slice()
    .sort((a, b) => a.faculty - b.faculty)
    .map((p) => htl.html`<tr>
      <td>F${p.faculty + 1}</td>
      <td>${studentNames[p.student]}</td>
      <td>${p.facultyRankOfStudent}</td>
      <td>${p.studentRankOfFaculty}</td>
    </tr>`);
  return htl.html`<table class="match-table">
    <thead>
      <tr><th>Faculty</th><th>Matched student</th><th>Faculty's rank of student (of ${nStudents})</th><th>Student's rank of faculty (of ${nFaculty})</th></tr>
    </thead>
    <tbody>${rows}</tbody>
  </table>`;
}

studentSummary = {
  const rowEls = studentNames.map((name, s) => {
    const cells = [htl.html`<td><strong>${name}</strong></td>`];
    for (const sem of semesters) {
      if (blocks[name] && blocks[name].includes(sem)) {
        cells.push(htl.html`<td><em>${blockReason[name][sem]}</em></td>`);
      } else {
        const facs = matching[sem].pairs
          .filter((p) => p.student === s)
          .map((p) => `F${p.faculty + 1}`)
          .join(", ");
        cells.push(htl.html`<td>${facs || "—"}</td>`);
      }
    }
    return htl.html`<tr>${cells}</tr>`;
  });
  return htl.html`<table class="match-table">
    <thead>
      <tr><th>Student</th><th>Fall</th><th>Spring</th></tr>
    </thead>
    <tbody>${rowEls}</tbody>
  </table>`;
